cmake_minimum_required(VERSION 3.20)
project(VIMEK VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

file(GLOB VIMEK_ENGINE_SOURCES CONFIGURE_DEPENDS Sources/VIMEK/engine/*.cpp)
add_library(vimek_engine STATIC ${VIMEK_ENGINE_SOURCES})
target_include_directories(vimek_engine PUBLIC Sources/VIMEK/engine)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(vimek_engine PUBLIC LINUX)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    option(VIMEK_BUILD_IBUS "Build the Linux IBus input method" ON)
    if(VIMEK_BUILD_IBUS)
        include(GNUInstallDirs)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(IBUS REQUIRED IMPORTED_TARGET ibus-1.0>=1.5.20)
        pkg_check_modules(CANBERRA REQUIRED IMPORTED_TARGET libcanberra)
        set(VIMEK_LINUX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Sources/VIMEK/linux")
        add_library(vimek_composer STATIC "${VIMEK_LINUX_DIR}/Composer.cpp")
        target_include_directories(vimek_composer PUBLIC "${VIMEK_LINUX_DIR}")
        target_link_libraries(vimek_composer PUBLIC vimek_engine)
        add_executable(ibus-engine-vimek "${VIMEK_LINUX_DIR}/IBusEngine.cpp")
        target_link_libraries(ibus-engine-vimek PRIVATE vimek_composer PkgConfig::IBUS PkgConfig::CANBERRA)
        target_compile_definitions(ibus-engine-vimek PRIVATE VIMEK_VERSION="${PROJECT_VERSION}"
            VIMEK_SETUP_PATH="${CMAKE_INSTALL_FULL_LIBEXECDIR}/vimek/vimek-settings")
        configure_file("${VIMEK_LINUX_DIR}/vimek.xml.in" "${CMAKE_CURRENT_BINARY_DIR}/vimek.xml" @ONLY)
        configure_file("${VIMEK_LINUX_DIR}/vimek.desktop.in" "${CMAKE_CURRENT_BINARY_DIR}/vimek.desktop" @ONLY)
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/schemas")
        configure_file("${VIMEK_LINUX_DIR}/org.vimek.settings.gschema.xml"
            "${CMAKE_CURRENT_BINARY_DIR}/schemas/org.vimek.settings.gschema.xml" COPYONLY)
        find_program(GLIB_COMPILE_SCHEMAS glib-compile-schemas REQUIRED)
        add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/schemas/gschemas.compiled"
            COMMAND "${GLIB_COMPILE_SCHEMAS}" --strict "${CMAKE_CURRENT_BINARY_DIR}/schemas"
            DEPENDS "${VIMEK_LINUX_DIR}/org.vimek.settings.gschema.xml")
        add_custom_target(vimek_schemas ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/schemas/gschemas.compiled")
        install(TARGETS ibus-engine-vimek RUNTIME DESTINATION "${CMAKE_INSTALL_LIBEXECDIR}/vimek")
        install(PROGRAMS "${VIMEK_LINUX_DIR}/settings.py" DESTINATION "${CMAKE_INSTALL_LIBEXECDIR}/vimek" RENAME vimek-settings)
        install(FILES "${CMAKE_CURRENT_BINARY_DIR}/vimek.xml" DESTINATION "${CMAKE_INSTALL_DATADIR}/ibus/component")
        install(FILES "${CMAKE_CURRENT_BINARY_DIR}/vimek.desktop" DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
        install(FILES "${VIMEK_LINUX_DIR}/org.vimek.settings.gschema.xml" DESTINATION "${CMAKE_INSTALL_DATADIR}/glib-2.0/schemas")
        install(FILES "${VIMEK_LINUX_DIR}/icons/vimek-v-symbolic.svg" "${VIMEK_LINUX_DIR}/icons/vimek-e-symbolic.svg"
            DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps")
        install(FILES LICENSE NOTICE.md README.md README.vi.md Linux_Build.md DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/vimek")
        set(VIMEK_RELEASE_VERSION "${PROJECT_VERSION}" CACHE STRING "Release/package version")
        set(CPACK_GENERATOR DEB)
        set(CPACK_PACKAGE_NAME vimek)
        set(CPACK_PACKAGE_VENDOR GrazT)
        set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "VIMEK Vietnamese input method for IBus")
        set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/dinhgia2106/VIMEK")
        set(CPACK_PACKAGE_VERSION "${VIMEK_RELEASE_VERSION}")
        set(CPACK_DEBIAN_PACKAGE_MAINTAINER GrazT)
        string(REPLACE "-" "~" CPACK_DEBIAN_PACKAGE_VERSION "${VIMEK_RELEASE_VERSION}")
        set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
        set(CPACK_DEBIAN_PACKAGE_DEPENDS "ibus (>= 1.5.20), ibus-gtk3, python3, python3-gi, gir1.2-gtk-3.0, libglib2.0-bin, libgtk-3-bin, hicolor-icon-theme, librsvg2-common")
        set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "libcanberra-pulse, sound-theme-freedesktop")
        if(CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
            set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE amd64)
        elseif(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
            set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
        else()
            message(FATAL_ERROR "Build the Linux package on amd64 or arm64")
        endif()
        set(CPACK_PACKAGE_FILE_NAME "VIMEK-${VIMEK_RELEASE_VERSION}-Linux-${CPACK_DEBIAN_PACKAGE_ARCHITECTURE}")
        set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${VIMEK_LINUX_DIR}/debian/postinst;${VIMEK_LINUX_DIR}/debian/postrm")
        set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION ON)
        include(CPack)
    endif()
endif()
if(MSVC)
    add_compile_options(/utf-8)
    target_compile_options(vimek_engine PRIVATE /utf-8)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE vimek_engine)
    add_test(NAME traditional_input COMMAND engine_tests)
    if(TARGET vimek_composer)
        add_executable(linux_composer_tests tests/linux_composer_tests.cpp)
        target_link_libraries(linux_composer_tests PRIVATE vimek_composer)
        add_test(NAME linux_composition COMMAND linux_composer_tests)
    endif()
endif()

if(WIN32)
    enable_language(RC)
    set(VIMEK_WIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Sources/VIMEK/windows/App")
    file(GLOB VIMEK_WIN_SOURCES CONFIGURE_DEPENDS "${VIMEK_WIN_DIR}/*.cpp")
    add_executable(VIMEK WIN32 ${VIMEK_WIN_SOURCES} "${VIMEK_WIN_DIR}/VIMEK.rc")
    target_include_directories(VIMEK PRIVATE "${VIMEK_WIN_DIR}")
    target_compile_definitions(VIMEK PRIVATE UNICODE _UNICODE VIMEK_PORTABLE_BUILD)
    set(VIMEK_WIN_LIBRARIES comctl32 comdlg32 shell32 ole32 uuid version uxtheme imm32 psapi gdiplus dwmapi gdi32)
    target_link_libraries(VIMEK PRIVATE vimek_engine ${VIMEK_WIN_LIBRARIES})
    if(MSVC)
        # The RC contains a complete manifest for both portable and MSVC builds.
        target_link_options(VIMEK PRIVATE /MANIFEST:NO)
    else()
        target_link_options(VIMEK PRIVATE -municode -static)
    endif()
    if(BUILD_TESTING)
        set(VIMEK_UI_TEST_SOURCES ${VIMEK_WIN_SOURCES})
        list(REMOVE_ITEM VIMEK_UI_TEST_SOURCES "${VIMEK_WIN_DIR}/main.cpp")
        add_executable(windows_ui_tests tests/windows_ui_tests.cpp ${VIMEK_UI_TEST_SOURCES} "${VIMEK_WIN_DIR}/VIMEK.rc")
        target_include_directories(windows_ui_tests PRIVATE "${VIMEK_WIN_DIR}")
        target_compile_definitions(windows_ui_tests PRIVATE UNICODE _UNICODE VIMEK_PORTABLE_BUILD)
        target_link_libraries(windows_ui_tests PRIVATE vimek_engine ${VIMEK_WIN_LIBRARIES})
        if(MSVC)
            target_link_options(windows_ui_tests PRIVATE /MANIFEST:NO)
        else()
            target_link_options(windows_ui_tests PRIVATE -static)
        endif()
        add_test(NAME windows_ui_dark COMMAND windows_ui_tests --preview-dark)
        add_test(NAME windows_ui_light COMMAND windows_ui_tests --preview-light)
    endif()
endif()
