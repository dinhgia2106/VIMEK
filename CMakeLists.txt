cmake_minimum_required(VERSION 3.20)
project(VIMEK VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

file(GLOB VIMEK_ENGINE_SOURCES CONFIGURE_DEPENDS Sources/VIMEK/engine/*.cpp)
add_library(vimek_engine STATIC ${VIMEK_ENGINE_SOURCES})
target_include_directories(vimek_engine PUBLIC Sources/VIMEK/engine)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(vimek_engine PUBLIC LINUX)
endif()
if(MSVC)
    add_compile_options(/utf-8)
    target_compile_options(vimek_engine PRIVATE /utf-8)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE vimek_engine)
    add_test(NAME traditional_input COMMAND engine_tests)
endif()

if(WIN32)
    enable_language(RC)
    set(VIMEK_WIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Sources/VIMEK/windows/App")
    file(GLOB VIMEK_WIN_SOURCES CONFIGURE_DEPENDS "${VIMEK_WIN_DIR}/*.cpp")
    add_executable(VIMEK WIN32 ${VIMEK_WIN_SOURCES} "${VIMEK_WIN_DIR}/VIMEK.rc")
    target_include_directories(VIMEK PRIVATE "${VIMEK_WIN_DIR}")
    target_compile_definitions(VIMEK PRIVATE UNICODE _UNICODE VIMEK_PORTABLE_BUILD)
    set(VIMEK_WIN_LIBRARIES comctl32 comdlg32 shell32 ole32 uuid version uxtheme imm32 psapi gdiplus dwmapi gdi32)
    target_link_libraries(VIMEK PRIVATE vimek_engine ${VIMEK_WIN_LIBRARIES})
    if(MSVC)
        # The RC contains a complete manifest for both portable and MSVC builds.
        target_link_options(VIMEK PRIVATE /MANIFEST:NO)
    else()
        target_link_options(VIMEK PRIVATE -municode -static)
    endif()
    if(BUILD_TESTING)
        set(VIMEK_UI_TEST_SOURCES ${VIMEK_WIN_SOURCES})
        list(REMOVE_ITEM VIMEK_UI_TEST_SOURCES "${VIMEK_WIN_DIR}/main.cpp")
        add_executable(windows_ui_tests tests/windows_ui_tests.cpp ${VIMEK_UI_TEST_SOURCES} "${VIMEK_WIN_DIR}/VIMEK.rc")
        target_include_directories(windows_ui_tests PRIVATE "${VIMEK_WIN_DIR}")
        target_compile_definitions(windows_ui_tests PRIVATE UNICODE _UNICODE VIMEK_PORTABLE_BUILD)
        target_link_libraries(windows_ui_tests PRIVATE vimek_engine ${VIMEK_WIN_LIBRARIES})
        if(MSVC)
            target_link_options(windows_ui_tests PRIVATE /MANIFEST:NO)
        else()
            target_link_options(windows_ui_tests PRIVATE -static)
        endif()
        add_test(NAME windows_ui_dark COMMAND windows_ui_tests --preview-dark)
        add_test(NAME windows_ui_light COMMAND windows_ui_tests --preview-light)
    endif()
endif()
